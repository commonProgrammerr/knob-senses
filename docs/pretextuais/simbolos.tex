\pretextualchapter{Lista de Símbolos}

\begingroup
\SingleSpacing
\begin{flushleft}
$P_{CA}$ -- Potência ativa em corrente alternada (kW)\par
$\hat{P}_{CA}$ -- Potência ativa estimada (kW)\par
$P_{\mathrm{CC}}$ -- Potência em corrente contínua (W)\par
$P_{\mathrm{STC}}$ -- Potência nominal nas condições padrão de teste (W)\par
$G_{\mathrm{POA}}$ -- Irradiância no plano dos módulos (W/m$^2$)\par
$G_{ghit}$ -- Irradiância global transposta (W/m$^2$)\par
$G_{\mathrm{STC}}$ -- Irradiância nas condições padrão de teste (1000~W/m$^2$)\par
$T_{amb}$ -- Temperatura ambiente ($^\circ$C)\par
$T_{\mathrm{cel}}$ -- Temperatura da célula ($^\circ$C)\par
$T_{mod}$ -- Temperatura estimada do módulo ($^\circ$C)\par
$T_{\mathrm{STC}}$ -- Temperatura nas condições padrão de teste ($25^\circ$C)\par
$\gamma$ -- Coeficiente térmico de potência (K$^{-1}$)\par
$\lambda$ -- Parâmetro da transformação Box--Cox\par
$\varepsilon$ -- Deslocamento aplicado ao alvo antes da transformação\par
$\delta$ -- Limiar da perda de Huber\par
$c_{\mathrm{cal}}$ -- Correção aditiva de viés (kW)\par
$y$ -- Valor medido\par
$\hat{y}$ -- Valor estimado\par
$e$ -- Erro de estimação ($e=y-\hat{y}$)\par
$R^2$ -- Coeficiente de determinação\par
$n$ -- Número de observações\par
\end{flushleft}
\endgroup
